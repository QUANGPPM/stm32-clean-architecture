cmake_minimum_required(VERSION 3.13)

# CMake workflow:
#   cmake -S . -B build      -> configure project
#   cmake --build build     -> build firmware
#   cmake --build build --target flash -> flash via OpenOCD
#   cmake --build build --target clean -> clean build outputs

# Use the project-specific ARM toolchain if no toolchain file is provided
if(NOT DEFINED CMAKE_TOOLCHAIN_FILE)
  set(CMAKE_TOOLCHAIN_FILE "${CMAKE_SOURCE_DIR}/cmake/arm-none-eabi.cmake" CACHE FILEPATH "Toolchain file")
endif()

project(f407_discovery C CXX ASM)

# C/C++ language standards for the firmware project
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)

# Generate compile_commands.json for clangd/IntelliSense
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

# Locate the CubeMX-generated folder; support both old and new naming styles
set(CUBEMX_ROOT ${CMAKE_SOURCE_DIR}/Boards/v1/CubeMX-Generated)
if(NOT EXISTS ${CUBEMX_ROOT})
  set(CUBEMX_ROOT ${CMAKE_SOURCE_DIR}/Boards/v1/CubeMX_Generated)
endif()
if(NOT EXISTS ${CUBEMX_ROOT})
  message(FATAL_ERROR "CubeMX generated folder not found. Expected Boards/v1/CubeMX-Generated or Boards/v1/CubeMX_Generated")
endif()
set(CUBEMX_CORE_INC_DIR ${CUBEMX_ROOT}/Core/Inc)
set(CUBEMX_CORE_SRC_DIR ${CUBEMX_ROOT}/Core/Src)

# MCU/CPU flags used by the ARM cross-compiler
set(MCU_FLAGS "-mthumb -mcpu=cortex-m4 -mfloat-abi=hard -mfpu=fpv4-sp-d16")
set(COMMON_FLAGS "-Og -g -ffunction-sections -fdata-sections ${MCU_FLAGS} -Wall")
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} ${COMMON_FLAGS}")
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -fno-exceptions -fno-rtti ${COMMON_FLAGS}")
set(CMAKE_ASM_FLAGS "${CMAKE_ASM_FLAGS} ${COMMON_FLAGS}")

# Linker script generated by CubeMX; tells the linker where the memory layout is
set(LINKER_SCRIPT ${CUBEMX_ROOT}/STM32F407VGTx_FLASH.ld)
if(EXISTS ${LINKER_SCRIPT})
  set(CMAKE_EXE_LINKER_FLAGS "${MCU_FLAGS} -Wl,--gc-sections -T${LINKER_SCRIPT} -specs=nosys.specs")
else()
  message(WARNING "Linker script not found: ${LINKER_SCRIPT}. Adjust CMakeLists or provide the file.")
endif()

# Collect all application/board/driver source files recursively
file(GLOB_RECURSE APP_SOURCES
  ${CMAKE_SOURCE_DIR}/Application/*.c
  ${CMAKE_SOURCE_DIR}/Application/*.cpp
  ${CMAKE_SOURCE_DIR}/Boards/**/*.c
  ${CMAKE_SOURCE_DIR}/Boards/**/*.cpp
  ${CMAKE_SOURCE_DIR}/Components/**/*.c
  ${CMAKE_SOURCE_DIR}/Components/**/*.cpp
  ${CMAKE_SOURCE_DIR}/Drivers/**/*.c
  ${CMAKE_SOURCE_DIR}/Drivers/**/*.cpp
)

# Add STM32 HAL driver sources; skip template files that are not meant to be built directly
file(GLOB HAL_SRCS ${CMAKE_SOURCE_DIR}/ThirdParty/STM32F4xx_HAL_Driver/Src/*.c)
list(FILTER HAL_SRCS EXCLUDE REGEX ".*/.*_template\\.c$")
if(HAL_SRCS)
  list(APPEND APP_SOURCES ${HAL_SRCS})
endif()

# Add FreeRTOS kernel and portable sources
file(GLOB_RECURSE FREERTOS_SRCS
  ${CMAKE_SOURCE_DIR}/ThirdParty/FreeRTOS/Source/*.c
  ${CMAKE_SOURCE_DIR}/ThirdParty/FreeRTOS/Source/portable/GCC/ARM_CM4F/*.c
)
set(CUBEMX_INCLUDE_DIR ${CUBEMX_CORE_INC_DIR})
if(NOT EXISTS ${CUBEMX_INCLUDE_DIR})
  set(CUBEMX_INCLUDE_DIR ${CUBEMX_ROOT}/Inc)
endif()
if(NOT EXISTS ${CUBEMX_INCLUDE_DIR})
  set(CUBEMX_INCLUDE_DIR "")
endif()

if(CUBEMX_INCLUDE_DIR)
  set(FREERTOS_CONFIG_FILE ${CUBEMX_INCLUDE_DIR}/FreeRTOSConfig.h)
  if(EXISTS ${FREERTOS_CONFIG_FILE})
    file(READ ${FREERTOS_CONFIG_FILE} FREERTOS_CONFIG)
  else()
    set(FREERTOS_CONFIG "")
  endif()
else()
  set(FREERTOS_CONFIG "")
endif()

if(FREERTOS_CONFIG MATCHES "USE_FreeRTOS_HEAP_1")
  set(FREERTOS_HEAP_SOURCE ${CMAKE_SOURCE_DIR}/ThirdParty/FreeRTOS/Source/portable/MemMang/heap_1.c)
elseif(FREERTOS_CONFIG MATCHES "USE_FreeRTOS_HEAP_2")
  set(FREERTOS_HEAP_SOURCE ${CMAKE_SOURCE_DIR}/ThirdParty/FreeRTOS/Source/portable/MemMang/heap_2.c)
elseif(FREERTOS_CONFIG MATCHES "USE_FreeRTOS_HEAP_3")
  set(FREERTOS_HEAP_SOURCE ${CMAKE_SOURCE_DIR}/ThirdParty/FreeRTOS/Source/portable/MemMang/heap_3.c)
elseif(FREERTOS_CONFIG MATCHES "USE_FreeRTOS_HEAP_4")
  set(FREERTOS_HEAP_SOURCE ${CMAKE_SOURCE_DIR}/ThirdParty/FreeRTOS/Source/portable/MemMang/heap_4.c)
elseif(FREERTOS_CONFIG MATCHES "USE_FreeRTOS_HEAP_5")
  set(FREERTOS_HEAP_SOURCE ${CMAKE_SOURCE_DIR}/ThirdParty/FreeRTOS/Source/portable/MemMang/heap_5.c)
else()
  message(WARNING "Unable to detect FreeRTOS heap implementation from ${FREERTOS_CONFIG_FILE}; defaulting to heap_4.c")
  set(FREERTOS_HEAP_SOURCE ${CMAKE_SOURCE_DIR}/ThirdParty/FreeRTOS/Source/portable/MemMang/heap_4.c)
endif()
list(FILTER FREERTOS_SRCS EXCLUDE REGEX ".*/portable/MemMang/heap_[1-5]\\.c$")
if(FREERTOS_SRCS)
  list(APPEND APP_SOURCES ${FREERTOS_SRCS})
endif()
list(APPEND APP_SOURCES ${FREERTOS_HEAP_SOURCE})

# Rename CubeMX's main() to cubemx_main so it does not collide with Application/main.cpp
set(CUBEMX_MAIN_C ${CUBEMX_CORE_SRC_DIR}/main.c)
if(EXISTS ${CUBEMX_MAIN_C})
  set_source_files_properties(${CUBEMX_MAIN_C} PROPERTIES COMPILE_DEFINITIONS "main=cubemx_main")
endif()

# Build the final firmware executable from all collected sources
set(STARTUP_SOURCES
  ${CUBEMX_ROOT}/startup_stm32f407xx.s
)

add_executable(firmware ${APP_SOURCES} ${STARTUP_SOURCES})

# Device-specific preprocessor definitions for ST HAL
target_compile_definitions(firmware PRIVATE STM32F407xx USE_HAL_DRIVER)

# Include directories needed by the firmware sources
target_include_directories(firmware PRIVATE
  ${CMAKE_SOURCE_DIR}/Application
  ${CMAKE_SOURCE_DIR}/Boards/v1
  ${CMAKE_SOURCE_DIR}/Components
  ${CMAKE_SOURCE_DIR}/Drivers
  ${CMAKE_SOURCE_DIR}/ThirdParty/CMSIS/Core/Include
  ${CMAKE_SOURCE_DIR}/ThirdParty/CMSIS/Device/ST/STM32F4xx/Include
  ${CMAKE_SOURCE_DIR}/ThirdParty/STM32F4xx_HAL_Driver/Inc
  ${CMAKE_SOURCE_DIR}/ThirdParty/FreeRTOS/Source/include
  ${CMAKE_SOURCE_DIR}/ThirdParty/FreeRTOS/Source/portable/GCC/ARM_CM4F
  ${CMAKE_SOURCE_DIR}/ThirdParty/FreeRTOS/Source/CMSIS_RTOS_V2
  ${CMAKE_SOURCE_DIR}/Application/tasks
)
if(CUBEMX_INCLUDE_DIR)
  target_include_directories(firmware PRIVATE ${CUBEMX_INCLUDE_DIR})
endif()

# Set target properties for the embedded ELF output
set_target_properties(firmware PROPERTIES
  LINKER_LANGUAGE C
  SUFFIX ".elf"
)

# Show firmware size after a successful build
add_custom_target(size ALL
  COMMAND arm-none-eabi-size -A ${CMAKE_CURRENT_BINARY_DIR}/firmware.elf
  DEPENDS firmware
  COMMENT "Showing binary sizes"
)

# Flash target for OpenOCD; adjust interface/target files if your debugger setup differs
add_custom_target(flash
  COMMAND ${CMAKE_COMMAND} -E echo "Flashing with OpenOCD..."
  COMMAND openocd -f interface/stlink.cfg -f target/stm32f4x.cfg -c "program ${CMAKE_CURRENT_BINARY_DIR}/firmware.elf verify reset exit"
  DEPENDS firmware
)

# Convenient distclean to remove build outputs
add_custom_target(distclean
  COMMAND ${CMAKE_COMMAND} -E rm -rf ${CMAKE_BINARY_DIR}/*
  COMMENT "Remove build directory contents"
)
